\subsection*{Pesquisa acadêmica}
O que a pesquisa diz sobre cobertura, mutação, testes instáveis, carga e
caos. Acesso (aberto ou pago) e links no apêndice~\ref{ap:pesquisa}.
\begin{itemize}
\pesquisa{paper:inozemtseva-coverage-2014}{Com o tamanho da suíte
  controlado, a cobertura quase não prevê a capacidade de achar defeitos.}
\pesquisa{paper:just-mutants-real-faults-2014}{Com defeitos reais de
  projetos Java (Defects4J), o mutation score acompanha a detecção de
  falhas melhor que a cobertura.}
\pesquisa{paper:jia-harman-mutation-survey-2011}{Panorama do teste de
  mutação: operadores, mutantes equivalentes, custo e ferramentas.}
\pesquisa{paper:zhu-coverage-adequacy-1997}{Referência sobre critérios de
  adequação: comandos, ramos, fluxo de dados e mutação.}
\pesquisa*{paper:delamaro-proteum-im-1997}{Ferramenta de mutação do
  ICMC-USP para teste de integração: a tradição brasileira em mutação.}
\pesquisa{paper:coles-pit-2016}{O paper do PIT, a ferramenta de mutação
  usada nesta aula.}
\pesquisa{paper:petrovic-mutation-google-2018}{Mutação em escala no
  Google, com mutantes mostrados no code review.}
\pesquisa{paper:claessen-quickcheck-2000}{O paper que criou os testes
  baseados em propriedades; o jqwik segue a mesma ideia.}
\pesquisa{paper:luo-flaky-tests-2014}{Causas de testes instáveis em
  projetos Apache: espera assíncrona, concorrência e ordem dos testes.}
\pesquisa{paper:parry-flaky-survey-2022}{Panorama sobre causas, detecção
  e correção de testes instáveis.}
\pesquisa{paper:eck-flaky-developers-2019}{Desenvolvedores da Mozilla
  classificam 200 testes instáveis.}
\pesquisa{paper:spadini-test-smells-2018}{Testes com smells mudam mais e
  falham mais, e o código que eles testam também.}
\pesquisa{paper:garousi-test-smells-2018}{Catálogo de test smells e de
  ferramentas que os detectam.}
\pesquisa{paper:jiang-load-testing-2015}{Como projetar, rodar e analisar
  testes de carga em sistemas grandes.}
\pesquisa{paper:basiri-chaos-engineering-2016}{Os princípios da engenharia
  do caos, escritos pelo time da Netflix e revisados por pares.}
\end{itemize}
