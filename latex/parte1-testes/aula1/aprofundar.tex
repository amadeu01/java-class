\subsection*{Pesquisa acadêmica}
O que a pesquisa diz sobre TDD e dublês. Acesso (aberto ou pago) e links no
apêndice~\ref{ap:pesquisa}.
\begin{itemize}
\pesquisa{paper:nagappan-tdd-industrial-2008}{Quatro times da Microsoft e
  da IBM adotaram TDD: menos defeitos antes do release, com 15 a 35\% a
  mais de tempo de desenvolvimento.}
\pesquisa{paper:fucci-dissection-tdd-2017}{Com 39 profissionais, o ganho
  veio de ciclos pequenos e uniformes, não de escrever o teste antes.}
\pesquisa{paper:rafique-tdd-metaanalise-2013}{Meta-análise de 27 estudos:
  efeito pequeno e positivo na qualidade, efeito incerto na produtividade.}
\pesquisa{paper:bissi-tdd-revisao-2016}{Revisão sistemática de autores
  brasileiros (UTFPR) que encontra ganho de qualidade externa com TDD.}
\pesquisa{paper:karac-what-we-know-tdd-2018}{Resumo curto da evidência
  sobre TDD, com foco em granularidade dos ciclos e refatoração.}
\pesquisa{paper:aniche-tdd-feedback-sbes-2012}{Do IME-USP: cataloga os
  sinais que os testes de unidade dão sobre o design das classes.}
\pesquisa{paper:aniche-tdd-class-design-jbcs-2015}{Entrevistas com 25
  desenvolvedores brasileiros: o TDD não desenha o código sozinho, mas dá
  segurança para refatorar e feedback sobre o design.}
\pesquisa{paper:silva-why-we-refactor-2016}{Da UFMG: por que as pessoas
  refatoram de verdade, o que conecta com o passo \emph{refactor}.}
\pesquisa{paper:spadini-to-mock-2017}{Em projetos Java com Mockito, as
  equipes mockam infraestrutura (banco, web services) e evitam mockar
  classes de domínio.}
\pesquisa{doc:mit-6031-testing}{Leitura aberta do MIT sobre test-first e
  partição do espaço de entrada; exemplos em TypeScript.}
\pesquisa{doc:cmu-17313}{Curso da CMU com aulas de testes de unidade,
  oráculos, fuzzing e testes baseados em propriedades.}
\end{itemize}
